<script>
// CONFIGURAÇÕES
const SEU_WHATSAPP = "5511999999999"; // Alterar para o seu número real
const DESCONTO_FIDELIDADE = 15;

// OPÇÕES DA ROLETA (Porcentagens e Cores)
const opcoesRoleta = [
{ desconto: 15, cor: '#d32f2f' },
{ desconto: 5, cor: '#1976d2' },
{ desconto: 10, cor: '#388e3c' },
{ desconto: 3, cor: '#f57c00' },
{ desconto: 12, cor: '#7b1fa2' },
{ desconto: 7, cor: '#0097a7' },
{ desconto: 1, cor: '#616161' },
{ desconto: 8, cor: '#e64a19' },
{ desconto: 13, cor: '#c2185b' },
{ desconto: 2, cor: '#455a64' }
];

let descontoGanho = 0;
let anguloAtual = 0;
let dadosCliente = {};

// DESENHAR ROLETA VIA CANVAS
const canvas = document.getElementById('roletaCanvas');
const ctx = canvas.getContext('2d');
const totalFatias = opcoesRoleta.length;
const anguloFatia = (2 * Math.PI) / totalFatias;

function desenharRoleta() {
const cx = canvas.width / 2;
const cy = canvas.height / 2;
const raio = canvas.width / 2 - 10;

ctx.clearRect(0, 0, canvas.width, canvas.height);

for (let i = 0; i < totalFatias; i++) {
const anguloInicio = i * anguloFatia;
const anguloFim = anguloInicio + anguloFatia;

// Desenhar Fatia
ctx.beginPath();
ctx.moveTo(cx, cy);
ctx.arc(cx, cy, raio, anguloInicio, anguloFim);
ctx.closePath();
ctx.fillStyle = opcoesRoleta[i].cor;
ctx.fill();
ctx.lineWidth = 4;
ctx.strokeStyle = '#ffffff';
ctx.stroke();

// Desenhar Texto (% Desconto)
ctx.save();
ctx.translate(cx, cy);
ctx.rotate(anguloInicio + anguloFatia / 2);
ctx.textAlign = "right";
ctx.fillStyle = "#ffffff";
ctx.font = "bold 28px Arial";
ctx.shadowColor = "rgba(0, 0, 0, 0.5)";
ctx.shadowBlur = 4;
ctx.fillText(opcoesRoleta[i].desconto + "%", raio - 25, 10);
ctx.restore();
}

// Círculo Central
ctx.beginPath();
ctx.arc(cx, cy, 30, 0, 2 * Math.PI);
ctx.fillStyle = '#ffffff';
ctx.fill();
ctx.lineWidth = 3;
ctx.strokeStyle = '#ddd';
ctx.stroke();
}

desenharRoleta();

// GIRAR ROLETA
function girarRoleta() {
const btnGirar = document.getElementById('btn-girar');
const resultado = document.getElementById('resultado-desconto');
const btnContinuar = document.getElementById('btn-continuar');

btnGirar.disabled = true;
btnGirar.style.opacity = "0.6";
btnGirar.textContent = "⏳ Sorteando...";

const indiceSorteado = Math.floor(Math.random() * totalFatias);
descontoGanho = opcoesRoleta[indiceSorteado].desconto;

// Cálculo exato para parar no ponteiro do topo (270 graus ou -90)
const grausPorFatia = 360 / totalFatias;
const anguloAlvo = 360 - (indiceSorteado * grausPorFatia) - (grausPorFatia / 2) - 90;

// Adiciona voltas completas (5 voltas = 1800 deg)
const voltas = 1800;
anguloAtual += voltas + (anguloAlvo - (anguloAtual % 360));

canvas.style.transform = `rotate(${anguloAtual}deg)`;

setTimeout(() => {
resultado.classList.remove('escondido');
resultado.innerHTML = `🎉 Você ganhou <strong>${descontoGanho}% DE DESCONTO!</strong>`;
btnContinuar.classList.remove('escondido');
btnGirar.style.display = 'none';
}, 4000);
}

// PASSOS DO FORMULÁRIO
function irPasso(n) {
if (n === 3) {
let nome = document.getElementById('cli-nome').value.trim();
let tel = document.getElementById('cli-tel').value.trim();
let end = document.getElementById('cli-endereco').value.trim();
if (!nome || !tel || !end) {
alert('⚠️ Preencha Nome, WhatsApp e Endereço para continuar!');
return;
}
salvarCadastro(nome, tel);
}

if (n === 4) {
if(!montarResumo()) return;
}

document.querySelectorAll('.passo').forEach(p => p.classList.remove('ativo'));
document.getElementById(`passo${n}`).classList.add('ativo');
}

function salvarCadastro(nome, tel) {
let chave = `cliente_${nome.replace(/\s/g, '_')}_${tel.replace(/\D/g, '')}`;
let cadastro = JSON.parse(localStorage.getItem(chave) || '{"pontos":0,"atendimentos":0}');

dadosCliente = {
nome, tel,
email: document.getElementById('cli-email').value.trim(),
endereco: document.getElementById('cli-endereco').value.trim(),
indicou: document.getElementById('cli-indicou').value.trim(),
chave
};

cadastro.atendimentos++;
let msgFidelidade = '';
if (cadastro.atendimentos >= 5) {
descontoGanho = Math.max(descontoGanho, DESCONTO_FIDELIDADE);
msgFidelidade = `\n💎 FIDELIDADE: ${cadastro.atendimentos}º Atendimento → ${DESCONTO_FIDELIDADE}% DE DESCONTO!`;
}

cadastro.ultimoAtendimento = new Date().toLocaleDateString('pt-BR');
localStorage.setItem(chave, JSON.stringify(cadastro));
dadosCliente.msgFidelidade = msgFidelidade;
dadosCliente.atendimentos = cadastro.atendimentos;
}

function montarResumo() {
let servicos = [];
let qtdTotal = 0;

const lista = [
{ id: 'inst', nome: 'Instalação' },
{ id: 'hig', nome: 'Higienização Completa' },
{ id: 'pre', nome: 'Pré-Higienização' },
{ id: 'gas', nome: 'Carga de Gás' },
{ id: 'rem', nome: 'Remoção' },
{ id: 'man', nome: 'Manutenção/Reparo' }
];

lista.forEach(s => {
if (document.getElementById(`serv-${s.id}`).checked) {
let qtd = document.getElementById(`qtd-${s.id}`).value || '1';
servicos.push(`• ${s.nome} (${qtd}x)`);
qtdTotal += parseInt(qtd);
}
});

if (servicos.length === 0) {
alert('⚠️ Selecione pelo menos um serviço para continuar!');
return false;
}

let resumo = document.getElementById('resumo-dados');
resumo.innerHTML = `
<p><strong>👤 Nome:</strong> ${dadosCliente.nome}</p>
<p><strong>📱 WhatsApp:</strong> ${dadosCliente.tel}</p>
<p><strong>📍 Endereço:</strong> ${dadosCliente.endereco}</p>
${dadosCliente.indicou ? `<p><strong>🤝 Indicado por:</strong> ${dadosCliente.indicou}</p>` : ''}
<hr style="margin:10px 0;border:none;border-top:1px dashed #ccc">
<p><strong>🔧 Serviços Selecionados:</strong></p>
<p>${servicos.join('<br>')}</p>
<p style="margin-top:6px;"><strong>📦 Qtd Total de Aparelhos:</strong> ${qtdTotal}</p>
<hr style="margin:10px 0;border:none;border-top:1px dashed #ccc">
<p><strong>🎁 Desconto Obtido:</strong> <span style="color:#2e7d32;font-weight:bold;">${descontoGanho}%</span></p>
${dadosCliente.msgFidelidade ? `<p style="color:#2e7d32;font-weight:bold">${dadosCliente.msgFidelidade}</p>` : ''}
`;

dadosCliente.servicos = servicos.join('\n');
dadosCliente.qtdTotal = qtdTotal;
return true;
}

function enviarPedido() {
const texto = `
🔔 *NOVO PEDIDO DE ATENDIMENTO!*

👤 *Cliente:* ${dadosCliente.nome}
📱 *WhatsApp:* ${dadosCliente.tel}
📧 *E-mail:* ${dadosCliente.email || 'Não informado'}
📍 *Endereço:* ${dadosCliente.endereco}
🤝 *Indicado por:* ${dadosCliente.indicou || 'Ninguém'}

🔧 *SERVIÇOS:*
${dadosCliente.servicos}

📦 *Total de Aparelhos:* ${dadosCliente.qtdTotal}
🎁 *Desconto da Roleta:* ${descontoGanho}%
📋 *Atendimento Nº:* ${dadosCliente.atendimentos}
${dadosCliente.msgFidelidade || ''}

📝 *Obs:* ${document.getElementById('obs-local').value || 'Nenhuma'}
`.trim();

const url = `https://wa.me/${SEU_WHATSAPP}?text=${encodeURIComponent(texto)}`;
window.open(url, '_blank');

document.querySelectorAll('.passo').forEach(p => p.classList.remove('ativo'));
document.getElementById('passo-fim').classList.add('ativo');
document.getElementById('desc-final').textContent = descontoGanho;
}
